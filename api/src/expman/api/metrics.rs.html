23/// How many points a chart gets by default.
24///
25/// Comfortably more than a wide chart has pixels, so downsampling is invisible,
26/// while bounding the JSON payload regardless of run length.
27const DEFAULT_MAX_POINTS: usize = 2000;
28
29#[derive(Deserialize)]
30pub struct MetricsQuery {
31    pub since_step: Option<u64>,
32    /// Cap on returned rows. Defaults to 2000; `0` means no cap.
33    pub max_points: Option<usize>,
34    /// Return every row. The escape hatch for export and analysis, where the
35    /// exact series matters and the caller is not a browser.
36    #[serde(default)]
37    pub full: bool,
38}
39
40pub async fn get_metrics(
41    State(state): State<AppState>,
42    Path((exp, run)): Path<(String, String)>,
43    Query(q): Query<MetricsQuery>,
44) -> impl IntoResponse {
45    let dir = run_dir(&state.base_dir, &exp, &run);
46    match storage::read_run_vectors_since(&dir, q.since_step) {
47        Ok(rows) => {
48            // A million-step run serialises to hundreds of MB of JSON and hangs
49            // the tab before it draws. Bound it unless the caller says otherwise.
50            let limit = if q.full {
51                0
52            } else {
53                q.max_points.unwrap_or(DEFAULT_MAX_POINTS)
54            };
55            let rows = if limit == 0 {
56                rows
57            } else {
58                storage::downsample_rows(rows, limit)
59            };
60            Json::<Vec<HashMap<String, serde_json::Value>>>(rows).into_response()
61        }
62        Err(e) => (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()).into_response(),
63    }
64}

66/// SSE endpoint: streams new vector metrics from vectors.parquet every 500ms.
67pub async fn stream_vectors(
68    State(state): State<AppState>,
69    Path((exp, run)): Path<(String, String)>,
70) -> Sse<impl tokio_stream::Stream<Item = Result<axum::response::sse::Event, Infallible>>> {
71    let dir = run_dir(&state.base_dir, &exp, &run);
72    let mut last_step: Option<u64> = None;
73
74    let interval = tokio::time::interval(Duration::from_millis(500));
75    let shutdown = state.shutdown_token.clone();
76    let stream = IntervalStream::new(interval)
77        .take_until(async move { shutdown.cancelled().await })
78        .map(move |_| {
79            let rows: Vec<HashMap<String, serde_json::Value>> =
80                storage::read_run_vectors_since(&dir, last_step).unwrap_or_default();
81            if let Some(row) = rows.last() {
82                let row: &HashMap<String, serde_json::Value> = row;
83                if let Some(step) = row.get("step").and_then(|v: &serde_json::Value| v.as_u64()) {
84                    last_step = Some(last_step.map_or(step, |ls| ls.max(step)));
85                }
86            }
87            let data: String = serde_json::to_string(&rows).unwrap_or_default();
88            Ok(axum::response::sse::Event::default().data(data))
89        });
90
91    Sse::new(stream).keep_alive(
92        axum::response::sse::KeepAlive::new()
93            .interval(Duration::from_secs(15))
94            .text("keep-alive"),
95    )
96}

103/// SSE endpoint: streams new lines from a log file (default: run.log) every 500ms.
104pub async fn stream_log(
105    State(state): State<AppState>,
106    Path((exp, run)): Path<(String, String)>,
107    Query(q): Query<LogQuery>,
108) -> Sse<impl tokio_stream::Stream<Item = Result<axum::response::sse::Event, Infallible>>> {
109    let filename = q.file.unwrap_or_else(|| "run.log".to_string());
110    let path = run_dir(&state.base_dir, &exp, &run).join(filename);
111    let mut last_pos: u64 = 0;
112
113    let interval = tokio::time::interval(Duration::from_millis(500));
114    let shutdown = state.shutdown_token.clone();
115    let stream = IntervalStream::new(interval)
116        .take_until(async move { shutdown.cancelled().await })
117        .map(move |_| {
118            let mut data = String::new();
119            if let Ok(file) = std::fs::File::open(&path) {
120                use std::io::{Read, Seek, SeekFrom};
121                let mut reader = std::io::BufReader::new(file);
122                let metadata = std::fs::metadata(&path).unwrap();
123                let len = metadata.len();
124
125                if len < last_pos {
126                    last_pos = 0;
127                }
128                if len > last_pos {
129                    let _ = reader.seek(SeekFrom::Start(last_pos));
130                    let _ = reader.read_to_string(&mut data);
131                    last_pos = len;
132                }
133            }
134            Ok(axum::response::sse::Event::default().data(data))
135        });
136
137    Sse::new(stream).keep_alive(
138        axum::response::sse::KeepAlive::new()
139            .interval(Duration::from_secs(15))
140            .text("keep-alive"),
141    )
142}

144pub async fn get_config(
145    State(state): State<AppState>,
146    Path((exp, run)): Path<(String, String)>,
147) -> impl IntoResponse {
148    let dir = run_dir(&state.base_dir, &exp, &run);
149    let path = dir.join("config.yaml");
150    match storage::load_yaml_value(&path) {
151        Ok(val) => match serde_json::to_value(&val) {
152            Ok(json_val) => Json(json_val).into_response(),
153            Err(e) => (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()).into_response(),
154        },
155        Err(e) => (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()).into_response(),
156    }
157}
