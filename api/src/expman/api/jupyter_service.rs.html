37///
38/// Documented in `docs/content/reference/cli.md`; kept here so the code and the
39/// docs can be checked against one list.
40pub const NOTEBOOK_PLACEHOLDERS: [&str; 5] =
41    ["run_dir", "run_name", "experiment", "store", "project"];

43/// Where a generated notebook's content comes from.
44///
45/// Resolution order is flag, then convention, then the built-in default — so a
46/// store with neither behaves exactly as it did before templating existed.
47#[derive(Debug, Clone, Default)]
48pub struct NotebookTemplateConfig {
49    /// `exp serve --notebook-template`, highest priority.
50    pub explicit: Option<PathBuf>,
51    /// The store root. `<base_dir>/.expman/notebook.ipynb` is the convention.
52    pub base_dir: PathBuf,
53}
54
55impl NotebookTemplateConfig {
56    /// The template to use, or `None` to use the built-in default.
57    ///
58    /// Absolute, because the path is recorded in the notebook it produces and a
59    /// relative one there would only mean anything next to the server's cwd.
60    pub fn resolve(&self) -> Option<PathBuf> {
61        if let Some(path) = &self.explicit {
62            if path.is_file() {
63                return Some(absolute(path));
64            }
65            // An explicit flag pointing at nothing is a typo worth naming, but
66            // not worth refusing to serve over: the convention below, or the
67            // built-in default, still produces a working notebook.
68            warn!(
69                "--notebook-template {} is not a file; ignoring it",
70                path.display()
71            );
72        }
73        let conventional = self.base_dir.join(CONVENTIONAL_TEMPLATE);
74        conventional.is_file().then(|| absolute(&conventional))
75    }
76}

78/// The values a template's placeholders are substituted with, for one run.
79#[derive(Debug, Clone)]
80pub struct NotebookContext {
81    /// Absolute path to the run directory — also where the notebook is written.
82    pub run_dir: PathBuf,
83    pub run_name: String,
84    pub experiment: String,
85    /// Absolute path to the store root (`base_dir`).
86    pub store: PathBuf,
87    /// The run's experiment's project; empty when the experiment is unassigned.
88    pub project: String,
89}
90
91impl NotebookContext {
92    /// Build the context for one run of one experiment.
93    ///
94    /// Paths are made absolute here rather than at substitution time, because a
95    /// template's whole purpose is to be runnable from the notebook's own
96    /// directory, where a relative `./experiments/...` would not resolve.
97    pub fn new(store: &Path, experiment: &str, run: &str, project: String) -> Self {
98        let store = absolute(store);
99        Self {
100            run_dir: store.join(experiment).join(run),
101            run_name: run.to_string(),
102            experiment: experiment.to_string(),
103            store,
104            project,
105        }
106    }
107
108    /// The placeholder → value bindings, in the documented order.
109    fn bindings(&self) -> [(&'static str, String); 5] {
110        [
111            ("run_dir", self.run_dir.display().to_string()),
112            ("run_name", self.run_name.clone()),
113            ("experiment", self.experiment.clone()),
114            ("store", self.store.display().to_string()),
115            ("project", self.project.clone()),
116        ]
117    }
118}

120/// Absolute form of `path`, without requiring it to exist.
121///
122/// `canonicalize` would resolve symlinks but fails on a run directory that has
123/// not been created yet, which is the common case on first launch.
124fn absolute(path: &Path) -> PathBuf {
125    std::path::absolute(path).unwrap_or_else(|_| path.to_path_buf())
126}

128/// Escape `value` for splicing inside a JSON string literal.
129///
130/// A template is `.ipynb` — JSON — and every placeholder sits inside a string
131/// literal, so a value containing `"`, `\`, or a control character has to be
132/// escaped or the substitution produces a corrupt notebook. Run directories
133/// really can contain such characters. `serde_json` quotes the value correctly;
134/// the quotes are then dropped because the surrounding literal supplies its own.
135fn json_escape(value: &str) -> String {
136    match serde_json::to_string(value) {
137        // Safe slice: `to_string` on a `&str` always yields `"…"`, and the
138        // quotes are ASCII, so 1..len-1 is on a char boundary.
139        Ok(quoted) => quoted[1..quoted.len() - 1].to_string(),
140        Err(_) => String::new(),
141    }
142}
143
144/// Substitute every `{{placeholder}}` in `template`, JSON-escaping the values.
145fn substitute(template: &str, ctx: &NotebookContext) -> String {
146    let mut out = template.to_string();
147    for (name, value) in ctx.bindings() {
148        out = out.replace(&format!("{{{{{}}}}}", name), &json_escape(&value));
149    }
150    out
151}

155/// The metadata block expman stamps into every notebook it writes, namespaced
156/// under `metadata` so nbformat and Jupyter both leave it alone.
157const EXPMAN_METADATA_KEY: &str = "expman";
158const TEMPLATE_HASH_KEY: &str = "template_hash";
159const CONTENT_HASH_KEY: &str = "content_hash";
160const TEMPLATE_KEY: &str = "template";
161const GENERATOR_KEY: &str = "generated_by";
162/// `metadata.expman.template` when no template was used.
163const BUILTIN_ORIGIN: &str = "builtin";
164
165/// A notebook rendered and ready to write, plus what it was rendered from.
166struct RenderedNotebook {
167    /// Parsed rather than kept as text, because the expman metadata block is
168    /// injected into it before writing.
169    value: serde_json::Value,
170    /// Hash of the *source* text, so both a template edit and a change to the
171    /// built-in default in a new expman version are detectable.
172    template_hash: String,
173    /// Recorded in the metadata: a template path, or `builtin`.
174    origin: String,
175}
176
177fn hash(text: &str) -> String {
178    use sha2::{Digest, Sha256};
179    format!("sha256:{:x}", Sha256::digest(text.as_bytes()))
180}
181
182/// Fingerprint a notebook for the "has the user edited this?" test.
183///
184/// Taken over the parsed JSON with `metadata.expman.content_hash` removed — the
185/// field cannot cover itself — and re-serialised, so the fingerprint does not
186/// depend on the file's whitespace. Executing cells *does* change it, which is
187/// intended: outputs are the user's work, not ours to overwrite.
188fn content_fingerprint(notebook: &serde_json::Value) -> String {
189    let mut probe = notebook.clone();
190    if let Some(expman) = probe
191        .pointer_mut("/metadata/expman")
192        .and_then(|v| v.as_object_mut())
193    {
194        expman.remove(CONTENT_HASH_KEY);
195    }
196    hash(&serde_json::to_string(&probe).unwrap_or_default())
197}
198
199/// Stamp `metadata.expman` into a rendered notebook and serialise it.
200fn seal(rendered: RenderedNotebook) -> String {
201    let RenderedNotebook {
202        mut value,
203        template_hash,
204        origin,
205    } = rendered;
206
207    // A valid .ipynb always has `metadata`, but a template is user-supplied, so
208    // create the object rather than discard the template over a missing key.
209    if !value.get("metadata").is_some_and(|m| m.is_object()) {
210        value["metadata"] = serde_json::json!({});
211    }
212    value["metadata"][EXPMAN_METADATA_KEY] = serde_json::json!({
213        GENERATOR_KEY: format!("expman {}", env!("CARGO_PKG_VERSION")),
214        TEMPLATE_KEY: origin,
215        TEMPLATE_HASH_KEY: template_hash,
216    });
217
218    // The content hash has to be computed with the field absent, then inserted.
219    let fingerprint = content_fingerprint(&value);
220    value["metadata"][EXPMAN_METADATA_KEY][CONTENT_HASH_KEY] = serde_json::json!(fingerprint);
221
222    serde_json::to_string_pretty(&value).unwrap_or_default()
223}

225/// What a launch should do about the notebook already on disk.
226enum Staleness {
227    /// Nothing there — write it.
228    Absent,
229    /// expman wrote it, it is untouched, and the template has moved on.
230    Regenerate,
231    /// Untouched and current.
232    Current,
233    /// Edited, or not written by expman. Never overwritten.
234    Keep { reason: &'static str },
235}
236
237/// Decide whether the notebook at `path` may be rewritten.
238async fn classify(path: &Path, template_hash: &str) -> Staleness {
239    let Ok(text) = tokio::fs::read_to_string(path).await else {
240        // Either absent, or present and unreadable. Treating an unreadable file
241        // as absent would overwrite it, so let the write attempt fail instead.
242        if path.exists() {
243            return Staleness::Keep {
244                reason: "it could not be read",
245            };
246        }
247        return Staleness::Absent;
248    };
249
250    let Ok(notebook) = serde_json::from_str::<serde_json::Value>(&text) else {
251        return Staleness::Keep {
252            reason: "it is not valid JSON",
253        };
254    };
255
256    let expman = notebook.pointer("/metadata/expman");
257    let recorded_content = expman
258        .and_then(|m| m.get(CONTENT_HASH_KEY))
259        .and_then(|h| h.as_str());
260    let recorded_template = expman
261        .and_then(|m| m.get(TEMPLATE_HASH_KEY))
262        .and_then(|h| h.as_str());
263
264    // A hand-made notebook, or one from a version that did not stamp metadata,
265    // counts as edited: we have no evidence it is ours to replace.
266    let (Some(recorded_content), Some(recorded_template)) = (recorded_content, recorded_template)
267    else {
268        return Staleness::Keep {
269            reason: "it carries no expman metadata, so expman did not write it",
270        };
271    };
272
273    if content_fingerprint(&notebook) != recorded_content {
274        return Staleness::Keep {
275            reason: "it has been edited since expman wrote it",
276        };
277    }
278    if recorded_template != template_hash {
279        return Staleness::Regenerate;
280    }
281    Staleness::Current
282}

284/// Render the notebook for one run: the project template if there is a usable
285/// one, else the built-in default.
286///
287/// A template that cannot be read, or whose substituted form is not valid JSON,
288/// is reported by path and the built-in default used instead. Writing a corrupt
289/// `.ipynb` would break the very tab the template exists to improve.
290async fn render_notebook(
291    template: &NotebookTemplateConfig,
292    ctx: &NotebookContext,
293    is_python: bool,
294) -> Result<RenderedNotebook, String> {
295    if let Some(path) = template.resolve() {
296        match tokio::fs::read_to_string(&path).await {
297            Ok(text) => {
298                let substituted = substitute(&text, ctx);
299                match serde_json::from_str(&substituted) {
300                    Ok(value) => {
301                        return Ok(RenderedNotebook {
302                            value,
303                            template_hash: hash(&text),
304                            origin: path.display().to_string(),
305                        })
306                    }
307                    Err(e) => error!(
308                        "Notebook template {} is not valid JSON after placeholder \
309                         substitution ({}); falling back to the built-in default",
310                        path.display(),
311                        e
312                    ),
313                }
314            }
315            Err(e) => error!(
316                "Could not read notebook template {}: {}; falling back to the \
317                 built-in default",
318                path.display(),
319                e
320            ),
321        }
322    }
323    builtin(generate_notebook_content(is_python))
324}

326/// Wrap built-in content as a `RenderedNotebook`.
327///
328/// Fallible rather than `expect`ing, because the built-in content is not a pure
329/// literal: the multi-run variant interpolates run names, which are directory
330/// names and so are only as well-behaved as the filesystem. A run named with a
331/// quote must fail this request, not panic the handler serving it.
332fn builtin(text: String) -> Result<RenderedNotebook, String> {
333    let value = serde_json::from_str(&text).map_err(|e| {
334        format!("expman generated an invalid built-in notebook ({e}); this is a bug")
335    })?;
336    Ok(RenderedNotebook {
337        value,
338        template_hash: hash(&text),
339        origin: BUILTIN_ORIGIN.to_string(),
340    })
341}

343/// Write `rendered` to `path` unless the file there must be left alone.
344///
345/// Returns `Ok(true)` when the file was written, `Ok(false)` when the existing
346/// one was kept.
347async fn write_notebook(path: &Path, rendered: RenderedNotebook) -> Result<bool, String> {
348    match classify(path, &rendered.template_hash).await {
349        Staleness::Current => return Ok(false),
350        Staleness::Keep { reason } => {
351            warn!(
352                "Leaving {} alone: {}. It may be stale relative to the current \
353                 notebook template; delete it to have expman regenerate it.",
354                path.display(),
355                reason
356            );
357            return Ok(false);
358        }
359        Staleness::Regenerate => info!(
360            "Regenerating {}: the notebook template changed and the file is unedited",
361            path.display()
362        ),
363        Staleness::Absent => {}
364    }
365
366    let content = seal(rendered);
367    if let Err(e) = tokio::fs::write(path, content).await {
368        error!("Failed to generate {}: {}", NOTEBOOK_FILE, e);
369        return Err(format!("Failed to generate {}: {}", NOTEBOOK_FILE, e));
370    }
371    Ok(true)
372}

376/// Generate the full `.ipynb` JSON content for a default interactive notebook.
377///
378/// This is the fallback used when a store supplies no template. For Python runs,
379/// produces 2 cells:
380///   1. Install dependencies (`pip install polars matplotlib`)
381///   2. Load and display metrics
382///
383/// For Rust runs, produces a single cell with a `polars` snippet.
384pub fn generate_notebook_content(is_python: bool) -> String {
385    let cells = if is_python {
386        r##"{
387   "cell_type": "code",
388   "execution_count": null,
389   "metadata": {},
390   "outputs": [],
391   "source": [
392    "# Install required dependencies into this environment\n",
393    "import sys\n",
394    "!pip --python {sys.executable} install polars matplotlib"
395   ]
396  },
397  {
398   "cell_type": "code",
399   "execution_count": null,
400   "metadata": {},
401   "outputs": [],
402   "source": [
403    "import polars as pl\n",
404    "import matplotlib.pyplot as plt\n",
405    "\n",
406    "# Load run vectors\n",
407    "vectors_path = 'vectors.parquet'\n",
408    "df = pl.read_parquet(vectors_path)\n",
409    "\n",
410    "# Display the latest metrics\n",
411    "df.tail()"
412   ]
413  }"##
414        .to_string()
415    } else {
416        let snippet = "use polars::prelude::*;\n\nfn main() -> Result<(), PolarsError> {\n    // Load run vectors\n    let mut file = std::fs::File::open(\"vectors.parquet\").unwrap();\n    let df = ParquetReader::new(&mut file).finish()?;\n\n    println!(\"{:?}\", df.tail(Some(5)));\n    Ok(())\n}";
417        format!(
418            r#"{{
419   "cell_type": "code",
420   "execution_count": null,
421   "metadata": {{}},
422   "outputs": [],
423   "source": [
424    "{}"
425   ]
426  }}"#,
427            snippet.replace('\n', "\\n").replace('"', "\\\"")
428        )
429    };
430
431    format!(
432        r#"{{
433 "cells": [
434  {}
435 ],
436 "metadata": {{}},
437 "nbformat": 4,
438 "nbformat_minor": 5
439}}"#,
440        cells
441    )
442}

444/// Write `interactive.ipynb` into the run directory.
445///
446/// Returns `Ok(true)` if the notebook was written, `Ok(false)` if the existing
447/// one was kept — either because it is already current, or because it has been
448/// edited (see `classify`).
449pub async fn generate_notebook(
450    ctx: &NotebookContext,
451    is_python: bool,
452    template: &NotebookTemplateConfig,
453) -> Result<bool, String> {
454    let path = ctx.run_dir.join(NOTEBOOK_FILE);
455    let rendered = render_notebook(template, ctx, is_python).await?;
456    write_notebook(&path, rendered).await
457}

459/// Generate the full `.ipynb` JSON content for a multi-run interactive notebook.
460///
461/// **Not templatable.** See `generate_multi_run_notebook`.
462pub fn generate_multi_run_notebook_content(is_python: bool, runs: &[String]) -> String {
463    // Run names are directory names, so they can hold anything the filesystem
464    // allows — including a quote, which would close the JSON string literal these
465    // snippets are spliced into and yield a notebook Jupyter cannot open. The
466    // Python branch splices straight into JSON so it escapes here; the Rust
467    // branch escapes its whole snippet at the end, so it must *not* (escaping
468    // twice would leave a lone backslash before the quote).
469    let cells = if is_python {
470        // Escaped once, here, because both snippets below are spliced straight
471        // into JSON string literals.
472        let escaped: Vec<String> = runs.iter().map(|run| json_escape(run)).collect();
473        let load_snippets = escaped.iter().map(|run| {
474            format!("df_{} = pl.read_parquet('{}/vectors.parquet').with_columns(pl.lit('{}').alias('run'))", run.replace('-', "_"), run, run)
475        }).collect::<Vec<_>>().join("\n");
476        let tail_snippets = escaped
477            .iter()
478            .map(|run| format!("df_{}.tail()", run.replace('-', "_")))
479            .collect::<Vec<_>>()
480            .join(", ");
481
482        format!(
483            r##"{{
484   "cell_type": "code",
485   "execution_count": null,
486   "metadata": {{}},
487   "outputs": [],
488   "source": [
489    "# Install required dependencies into this environment\n",
490    "import sys\n",
491    "!pip --python {{sys.executable}} install polars matplotlib"
492   ]
493  }},
494  {{
495   "cell_type": "code",
496   "execution_count": null,
497   "metadata": {{}},
498   "outputs": [],
499   "source": [
500    "import polars as pl\n",
501    "import matplotlib.pyplot as plt\n",
502    "\n",
503    "# Load run vectors\n",
504    "{}\n",
505    "\n",
506    "# Display the latest metrics\n",
507    "{}"
508   ]
509  }}"##,
510            load_snippets.replace('\n', "\\n"),
511            tail_snippets.replace('\n', "\\n")
512        )
513    } else {
514        let load_snippets = runs.iter().map(|run| {
515            format!("    let df_{} = ParquetReader::new(&mut std::fs::File::open(\"{}/vectors.parquet\").unwrap()).finish()?;\n    // Note: To add a 'run' column in rust polars you would typically use lit(\"{}\") in a select/with_columns, \n    // but for simplicity here we just load them.", run.replace('-', "_"), run, run)
516        }).collect::<Vec<_>>().join("\n");
517
518        let snippet = format!("use polars::prelude::*;\n\nfn main() -> Result<(), PolarsError> {{\n    // Load run vectors\n{}\n    Ok(())\n}}", load_snippets);
519        format!(
520            r#"{{
521   "cell_type": "code",
522   "execution_count": null,
523   "metadata": {{}},
524   "outputs": [],
525   "source": [
526    "{}"
527   ]
528  }}"#,
529            json_escape(&snippet)
530        )
531    };
532
533    format!(
534        r#"{{
535 "cells": [
536  {}
537 ],
538 "metadata": {{}},
539 "nbformat": 4,
540 "nbformat_minor": 5
541}}"#,
542        cells
543    )
544}

546/// Write the multi-run `interactive.ipynb` into the experiment directory.
547///
548/// **Deliberately always the built-in default: `--notebook-template` does not
549/// apply here.** A multi-run notebook has no single run, so `{{run_dir}}`,
550/// `{{run_name}}` and `{{project}}` have no value to take, and a useful shared
551/// template would need a different placeholder set (a `{{runs}}` list, and a
552/// rule for rendering it) — a separate design, not a wider version of this one.
553/// Recorded in `docs/content/reference/cli.md`.
554///
555/// The staleness contract *is* shared: the built-in content includes the run
556/// names, so adding a run refreshes an unedited notebook, and an edited one is
557/// still never overwritten.
558pub async fn generate_multi_run_notebook(
559    exp_dir: &Path,
560    is_python: bool,
561    runs: &[String],
562) -> Result<bool, String> {
563    let path = exp_dir.join(NOTEBOOK_FILE);
564    let rendered = builtin(generate_multi_run_notebook_content(is_python, runs))?;
565    write_notebook(&path, rendered).await
566}

570/// The command line that launches Jupyter, split into program and leading args.
571///
572/// Configured by `exp serve --jupyter-command`; defaults to a bare `jupyter`.
573///
574/// **Why a command line rather than a program name.** The kernel a notebook gets
575/// is the interpreter Jupyter itself runs under: launched from inside a
576/// project's virtualenv, Jupyter's built-in `python3` kernel *is* that venv's
577/// interpreter, so `import <project package>` works with no `ipykernel install`
578/// and no kernelspec name to keep in sync between expman and the project. That
579/// is the whole point of the flag — `--jupyter-command 'uv run --extra nb
580/// jupyter'` — and the reason expman deliberately does **not** register a global
581/// kernel or write a named kernelspec into the generated notebook. Either of
582/// those would mutate the user's `~/.local/share/jupyter` and then have to be
583/// kept correct forever.
584#[derive(Debug, Clone)]
585pub struct JupyterCommand {
586    /// The configured line, verbatim, for error messages.
587    raw: String,
588    program: String,
589    leading_args: Vec<String>,
590}
591
592impl JupyterCommand {
593    /// Split a configured command line into a program and its leading arguments.
594    ///
595    /// Splitting is POSIX-shell-like (via `shlex`), so a quoted path containing a
596    /// space survives. No shell is involved: the program is executed directly, so
597    /// pipes, redirections and variable expansion are not available.
598    pub fn parse(raw: &str) -> Result<Self, String> {
599        let words = shlex::split(raw).ok_or_else(|| {
600            format!(
601                "--jupyter-command {:?} is not a parsable command line \
602                 (unbalanced quote or trailing backslash?)",
603                raw
604            )
605        })?;
606        let mut words = words.into_iter();
607        let program = words
608            .next()
609            .ok_or_else(|| "--jupyter-command must name a program".to_string())?;
610        Ok(Self {
611            raw: raw.to_string(),
612            program,
613            leading_args: words.collect(),
614        })
615    }
616
617    /// A `Command` for the configured line, ready for the Jupyter arguments.
618    ///
619    /// The configured words go first, so `uv run --extra nb jupyter` followed by
620    /// `notebook …` composes into the command line a user would type by hand.
621    fn base_command(&self) -> tokio::process::Command {
622        let mut cmd = tokio::process::Command::new(&self.program);
623        cmd.args(&self.leading_args);
624        cmd
625    }
626
627    /// The configured line as given, for messages the user has to act on.
628    pub fn as_str(&self) -> &str {
629        &self.raw

656/// Detect the best available interactive Python backend in the user's environment.
657///
658/// Checks the configured Jupyter command, then `python3`.
659pub async fn detect_backend(jupyter: &JupyterCommand) -> InteractiveBackend {
660    // Check jupyter
661    match jupyter
662        .base_command()
663        .args(["notebook", "--version"])
664        .output()
665        .await
666    {
667        Ok(output) => {
668            info!(
669                "`{} notebook --version`: status={}, stdout={}, stderr={}",
670                jupyter.as_str(),
671                output.status,
672                String::from_utf8_lossy(&output.stdout).trim(),
673                String::from_utf8_lossy(&output.stderr).trim()
674            );
675            if output.status.success() {
676                return InteractiveBackend::Jupyter;
677            }
678        }
679        Err(e) => {
680            info!("`{}` not found: {}", jupyter.as_str(), e);
681        }
682    }
683
684    // Check python3
685    match tokio::process::Command::new("python3")
686        .arg("--version")
687        .output()
688        .await
689    {
690        Ok(output) => {
691            info!(
692                "python3 --version: status={}, stdout={}",
693                output.status,
694                String::from_utf8_lossy(&output.stdout).trim()
695            );
696            if output.status.success() {
697                return InteractiveBackend::Python;
698            }
699        }
700        Err(e) => {
701            info!("python3 not found: {}", e);
702        }
703    }
704
705    InteractiveBackend::None
706}

708/// Thread-safe manager for spawning and stopping Jupyter Notebooks.
709///
710/// When Jupyter is available in the user's environment, this manager spawns
711/// per-run Jupyter instances. When only ipython/python is available, the
712/// frontend shows notebook content with copy-paste guidance instead.
713#[derive(Clone, Default)]
714pub struct JupyterManager {
715    instances: Arc<Mutex<HashMap<String, JupyterInstance>>>,
716    /// How to launch Jupyter (`exp serve --jupyter-command`).
717    command: JupyterCommand,
718    /// Where generated notebooks come from (`exp serve --notebook-template`).
719    template: NotebookTemplateConfig,
720}
721
722impl JupyterManager {
723    pub fn new() -> Self {
724        Self::default()
725    }
726
727    /// A manager that honours the server's Jupyter configuration.
728    pub fn with_config(command: JupyterCommand, template: NotebookTemplateConfig) -> Self {
729        Self {
730            instances: Arc::default(),
731            command,
732            template,
733        }
734    }
735
736    /// The configured Jupyter command, for backend detection.
737    pub fn command(&self) -> &JupyterCommand {
738        &self.command
739    }
740
741    /// The configured notebook template source.
742    pub fn template(&self) -> &NotebookTemplateConfig {
743        &self.template
744    }
745
746    /// Finds an available TCP port.
747    fn get_available_port() -> Option<u16> {
748        (8888..9999).find(|port| TcpListener::bind(("127.0.0.1", *port)).is_ok())
749    }
750
751    /// Launch a Jupyter server in `cwd` on `port`, failing if it dies at once.
752    async fn spawn_server(&self, cwd: &Path, port: u16) -> Result<Child, String> {
753        let mut child = self
754            .command
755            .base_command()
756            .args(notebook_args(port))
757            .current_dir(cwd)
758            .spawn()
759            .map_err(|e| {
760                format!(
761                    "Failed to start Jupyter with the configured command `{}`: {}. \
762                     Point `exp serve --jupyter-command` at a command line that \
763                     launches Jupyter (for example `uv run --extra nb jupyter`).",
764                    self.command.as_str(),
765                    e
766                )
767            })?;
768
769        // Small wait to ensure it hasn't instantly crashed
770        tokio::time::sleep(tokio::time::Duration::from_millis(500)).await;
771        if let Ok(Some(status)) = child.try_wait() {
772            return Err(format!(
773                "Jupyter (`{}`) crashed immediately with status {}",
774                self.command.as_str(),
775                status
776            ));
777        }
778        Ok(child)
779    }
780
781    /// Spawns a Jupyter Notebook for a given run.
782    pub async fn spawn(&self, ctx: &NotebookContext, is_python: bool) -> Result<u16, String> {
783        let key = format!("{}:{}", ctx.experiment, ctx.run_name);
784
785        // Already running?
786        {
787            let instances = self.instances.lock().unwrap();
788            if let Some(instance) = instances.get(&key) {
789                return Ok(instance.port);
790            }
791        }
792
793        let port = Self::get_available_port()
794            .ok_or_else(|| "No available ports for Jupyter".to_string())?;
795
796        // Generate the notebook, or refresh it if the template has moved on
797        generate_notebook(ctx, is_python, &self.template).await?;
798
799        info!("Spawning Jupyter Notebook for {} on port {}", key, port);
800        let child = self.spawn_server(&ctx.run_dir, port).await?;
801
802        let mut instances = self.instances.lock().unwrap();
803        instances.insert(
804            key,
805            JupyterInstance {
806                port,
807                process: child,
808            },
809        );
810
811        Ok(port)
812    }

814    /// Spawns a multi-run Jupyter Notebook in the experiment directory.
815    pub async fn spawn_multi(
816        &self,
817        exp: &str,
818        exp_dir: PathBuf,
819        is_python: bool,
820        runs: &[String],
821    ) -> Result<u16, String> {
822        let key = format!("{}:__multi__", exp);
823
824        // Already running?
825        {
826            let instances = self.instances.lock().unwrap();
827            if let Some(instance) = instances.get(&key) {
828                return Ok(instance.port);
829            }
830        }
831
832        let port = Self::get_available_port()
833            .ok_or_else(|| "No available ports for Jupyter".to_string())?;

835        // Generate notebook if it doesn't exist
836        generate_multi_run_notebook(&exp_dir, is_python, runs).await?;
837
838        info!(
839            "Spawning multi-run Jupyter Notebook for {} on port {}",
840            exp, port

845        instances.insert(

850            },

856    /// Returns the port if the notebook is running.
857    pub fn status(&self, exp: &str, run: &str) -> Option<u16> {
858        let key = format!("{}:{}", exp, run);
859        let mut instances = self.instances.lock().unwrap();
860
861        if let Some(instance) = instances.get_mut(&key) {
862            match instance.process.try_wait() {
863                Ok(Some(_)) => { /* exited */ }
864                Ok(None) => return Some(instance.port),
865                Err(_) => { /* error polling */ }
866            }
867        }
868
869        instances.remove(&key);
870        None
871    }
872
873    /// Stops a running Jupyter instance.
874    pub async fn stop(&self, exp: &str, run: &str) -> Result<(), String> {
875        let key = format!("{}:{}", exp, run);
876        let mut instance = {
877            let mut instances = self.instances.lock().unwrap();
878            instances.remove(&key)
879        };
880
881        if let Some(mut inst) = instance.take() {
882            info!("Shutting down Jupyter Notebook for {}", key);
883            let _ = inst.process.kill().await;
884            let _ = inst.process.wait().await;
885        }
886
887        Ok(())
888    }
889
890    /// Kill all notebooks (e.g., on server shutdown).
891    pub async fn shutdown_all(&self) {
892        let all: Vec<_> = {
893            let mut instances = self.instances.lock().unwrap();
894            instances.drain().map(|(_, inst)| inst).collect()
895        };
896
897        for mut inst in all {
898            let _ = inst.process.kill().await;
899            let _ = tokio::time::timeout(tokio::time::Duration::from_secs(5), inst.process.wait())
900                .await;
901        }
902    }
