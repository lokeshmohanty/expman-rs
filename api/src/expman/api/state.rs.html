1//! Shared application state for the Axum server.
2
3use std::path::PathBuf;
4use std::sync::Arc;
5use tokio_util::sync::CancellationToken;
6use tracing::error;
7
8use super::jupyter_service::{
9    JupyterCommand, JupyterManager, NotebookTemplateConfig, DEFAULT_JUPYTER_COMMAND,
10};
11
12#[derive(Clone)]
13pub struct AppState {
14    pub base_dir: Arc<PathBuf>,
15    pub jupyter: super::jupyter_service::JupyterManager,
16    pub tensorboard: super::tensorboard_service::TensorBoardManager,
17    pub shutdown_token: CancellationToken,
18    /// Whether SSE streaming is enabled. Reported by `/api/config`.
19    pub live_mode: bool,
20    /// Whether every mutating request is refused. Reported by `/api/config` so
21    /// the frontend can hide controls rather than offering writes that 403.
22    pub read_only: bool,
23}
24
25impl AppState {
26    pub fn new(base_dir: PathBuf) -> Self {
27        Self::from_config(&ServerConfig {
28            base_dir,
29            ..Default::default()
30        })
31    }
32
33    pub fn from_config(config: &ServerConfig) -> Self {
34        // `exp serve` rejects an unparsable command up front, so this branch only
35        // guards a library caller that built a `ServerConfig` by hand. Falling
36        // back beats refusing to serve: every other tab still works.
37        let jupyter_command = JupyterCommand::parse(&config.jupyter_command).unwrap_or_else(|e| {
38            error!("{e}; falling back to `{DEFAULT_JUPYTER_COMMAND}`");
39            JupyterCommand::default()
40        });
41        let template = NotebookTemplateConfig {
42            explicit: config.notebook_template.clone(),
43            base_dir: config.base_dir.clone(),
44        };
45
46        Self {
47            base_dir: Arc::new(config.base_dir.clone()),
48            jupyter: JupyterManager::with_config(jupyter_command, template),
49            tensorboard: super::tensorboard_service::TensorBoardManager::new(),
50            shutdown_token: CancellationToken::new(),
51            live_mode: config.live_mode,
52            read_only: config.read_only,
53        }
54    }
55}
56
57/// Configuration for the web server.
58#[derive(Debug, Clone)]
59pub struct ServerConfig {
60    pub base_dir: PathBuf,
61    pub host: String,
62    pub port: u16,
63    pub live_mode: bool,
64    /// Refuse every mutating request, so a store can be shared read-only.
65    pub read_only: bool,
66    /// A project-supplied `.ipynb` to generate interactive notebooks from
67    /// (`--notebook-template`). `None` falls back to
68    /// `<base_dir>/.expman/notebook.ipynb`, then to the built-in default.
69    pub notebook_template: Option<PathBuf>,
70    /// The command line that launches Jupyter (`--jupyter-command`), e.g.
71    /// `uv run --extra nb jupyter` to get the project's virtualenv as the kernel.
72    pub jupyter_command: String,
73}
74
75impl Default for ServerConfig {
76    fn default() -> Self {
77        Self {
78            base_dir: PathBuf::from("experiments"),
79            host: "127.0.0.1".to_string(),
80            port: 8000,
81            live_mode: true,
82            read_only: false,
83            notebook_template: None,
84            jupyter_command: DEFAULT_JUPYTER_COMMAND.to_string(),
85        }
86    }
87}
