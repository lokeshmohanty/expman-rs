1//! System metrics: hardware utilisation sampled alongside the training metrics.
2//!
3//! Collection is by **subprocess probe** — we shell out to the vendor's own CLI
4//! (`nvidia-smi`, `rocm-smi`, `tpu-info`) rather than linking NVML or ROCm. That
5//! trades a few milliseconds per sample for three things worth more: no
6//! build-time dependency on a vendor SDK, no runtime `dlopen` that fails
7//! differently on every cluster image, and — most importantly — a user can add
8//! a probe for hardware we have never heard of by writing a config entry
9//! instead of a patch.
10//!
11//! A probe whose binary is absent is skipped silently and permanently. A laptop
12//! with no GPU logs CPU and memory and says nothing about it.
13//!
14//! ## Parsing strategy
15//!
16//! Only two parsers, because inventing a schema for each vendor is how this
17//! kind of code rots:
18//!
19//! - **`nvidia-smi`** uses `--format=csv,noheader,nounits` with an explicit
20//!   `--query-gpu` column list, so the output shape is one we requested.
21//! - **Everything else JSON** is flattened by extracting *every numeric leaf*
22//!   into a dotted key. This is deliberately schema-agnostic: it works for
23//!   `rocm-smi --json`, for `tpu-info --json`, and for whatever a user's own
24//!   tool emits, and it does not break when a vendor adds a field.
25

34/// How to interpret a probe's stdout.
35#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
36#[serde(rename_all = "snake_case")]
37pub enum ProbeFormat {
38    /// `nvidia-smi --format=csv,noheader,nounits`, one row per device.
39    NvidiaCsv,
40    /// Any JSON. Every numeric leaf becomes a metric, keyed by its dotted path.
41    Json,
42    /// `key=value` or `key: value` lines, one per line.
43    KeyValue,
44    /// Markdown-style pipe tables: first column identifies the device, the
45    /// remaining column headers name the metrics. What `tpu-info` emits.
46    PipeTable,
47}

49/// A configured probe. Serializable so it can come from a config file.
50#[derive(Debug, Clone, Serialize, Deserialize)]
51pub struct ProbeSpec {
52    /// Metric key prefix, e.g. `gpu` → `gpu.0.util_pct`.
53    pub prefix: String,
54    /// Program to run. Skipped entirely if not found on PATH.
55    pub command: String,
56    #[serde(default)]
57    pub args: Vec<String>,
58    pub format: ProbeFormat,
59    /// Column names for `NvidiaCsv`, positionally matched to the CSV fields.
60    #[serde(default)]
61    pub columns: Vec<String>,
62}

64impl ProbeSpec {
65    /// NVIDIA GPUs. The query list fixes the column order, so parsing is not
66    /// guessing at a format — it is reading back what we asked for.
67    ///
68    /// Verified end to end against a 2× RTX A6000 host on 2026-07-29; the test
69    /// `nvidia_probe_matches_real_a6000_output` pins that exact output.
70    pub fn nvidia() -> Self {
71        Self {
72            prefix: "gpu".to_string(),
73            command: "nvidia-smi".to_string(),
74            args: vec![
75                "--query-gpu=index,utilization.gpu,utilization.memory,memory.used,memory.total,temperature.gpu,power.draw".to_string(),
76                "--format=csv,noheader,nounits".to_string(),
77            ],
78            format: ProbeFormat::NvidiaCsv,
79            columns: vec![
80                "index".to_string(),
81                "util_pct".to_string(),
82                "mem_util_pct".to_string(),
83                "mem_used_mb".to_string(),
84                "mem_total_mb".to_string(),
85                "temp_c".to_string(),
86                "power_w".to_string(),
87            ],
88        }
89    }

91    /// AMD GPUs via `rocm-smi --json`, read through the generic JSON flattener.
92    pub fn rocm() -> Self {
93        Self {
94            prefix: "gpu".to_string(),
95            command: "rocm-smi".to_string(),
96            args: vec![
97                "--showuse".to_string(),
98                "--showmemuse".to_string(),
99                "--showpower".to_string(),
100                "--showtemp".to_string(),
101                "--json".to_string(),
102            ],
103            format: ProbeFormat::Json,
104            columns: vec![],
105        }
106    }
107
108    /// Google TPUs via the `tpu-info` CLI.
109    ///
110    /// Verified against a v6e host (`tpu-info` 0.0.21, libtpu 0.0.21) on
111    /// 2026-07-29 — the earlier `--json` guess was simply wrong: `tpu-info` has
112    /// **no JSON output**, and prints Rich-formatted pipe tables, one per
113    /// requested metric.
114    ///
115    /// Only the three metrics that describe utilisation are requested. The tool
116    /// exposes ~40, most of them checkpoint and gRPC timings that belong in a
117    /// profile rather than in a 15-second sample.
118    ///
119    /// > `hbm_usage` and `duty_cycle_percent` read `N/A` unless a framework has
120    /// > the TPU open — libtpu publishes them only then. That is not an error and
121    /// > not worth warning about; the values simply appear once training starts.
122    /// > `tensorcore_utilization` reports even when idle, which is what was used
123    /// > to confirm numeric parsing against the live host.
124    ///
125    /// ## Why not talk to libtpu directly?
126    ///
127    /// Investigated on the v6e host, 2026-07-29. `tpu_info` reaches libtpu over
128    /// gRPC at `localhost:8431` (`RuntimeMetricServiceStub`, local channel
129    /// credentials), so a native client is conceivable. It was rejected:
130    ///
131    /// 1. **`ss -ltn` shows nothing on 8431 while the TPU is idle.** libtpu
132    ///    starts that server *inside* the training process. A direct client
133    ///    would get connection-refused exactly as often as `tpu-info` returns
134    ///    `N/A`, so it fixes nothing about availability — the thing worth fixing.
135    /// 2. **No `.proto` source ships** — only compiled `_pb2.py`. Using tonic
136    ///    means vendoring a reconstructed, Google-internal, unversioned schema
137    ///    that a libtpu update can silently break.
138    /// 3. `grpc.local_channel_credentials()` has no tonic equivalent.
139    /// 4. The saving is small: `tpu-info` costs ~391ms (of which ~100ms is
140    ///    Python import), against a default 15s interval — **~2.6% of one core**
141    ///    on a background thread that never touches the training loop.
142    ///
143    /// Revisit only if sub-second sampling is ever needed, which profiling —
144    /// not this sampler — is the right tool for.
145    pub fn tpu() -> Self {
146        Self {
147            prefix: "tpu".to_string(),
148            command: "tpu-info".to_string(),
149            args: vec![
150                "--metric".to_string(),
151                "duty_cycle_percent".to_string(),
152                "--metric".to_string(),
153                "hbm_usage".to_string(),
154                "--metric".to_string(),
155                "tensorcore_utilization".to_string(),
156            ],
157            format: ProbeFormat::PipeTable,
158            columns: vec![],
159        }
160    }
161
162    /// The probes tried by default. Absent binaries cost one PATH lookup each.
163    pub fn defaults() -> Vec<Self> {
164        vec![Self::nvidia(), Self::rocm(), Self::tpu()]
165    }
166
167    fn is_available(&self) -> bool {
168        which(&self.command).is_some()
169    }
170
171    fn sample(&self) -> HashMap<String, MetricValue> {
172        let output = match Command::new(&self.command).args(&self.args).output() {
173            Ok(o) if o.status.success() => o.stdout,
174            // A probe that errors once will usually error every time; we simply
175            // return nothing rather than filling the log with vendor noise.
176            _ => return HashMap::new(),
177        };
178        let text = String::from_utf8_lossy(&output);
179        match self.format {
180            ProbeFormat::NvidiaCsv => parse_nvidia_csv(&text, &self.prefix, &self.columns),
181            ProbeFormat::Json => serde_json::from_str::<serde_json::Value>(&text)
182                .map(|v| flatten_json_numbers(&v, &self.prefix))
183                .unwrap_or_default(),
184            ProbeFormat::KeyValue => parse_key_value(&text, &self.prefix),
185            ProbeFormat::PipeTable => parse_pipe_tables(&text, &self.prefix),
186        }
187    }
188}

190/// Minimal PATH lookup — avoids a dependency for six lines.
191fn which(program: &str) -> Option<std::path::PathBuf> {
192    if program.contains('/') {
193        let p = Path::new(program);
194        return p.is_file().then(|| p.to_path_buf());
195    }
196    std::env::var_os("PATH").and_then(|paths| {
197        std::env::split_paths(&paths).find_map(|dir| {
198            let candidate = dir.join(program);
199            candidate.is_file().then_some(candidate)
200        })
201    })
202}

204/// One row per device; `index` (if present) becomes part of the key.
205fn parse_nvidia_csv(text: &str, prefix: &str, columns: &[String]) -> HashMap<String, MetricValue> {
206    let mut out = HashMap::new();
207    for (row_idx, line) in text.lines().filter(|l| !l.trim().is_empty()).enumerate() {
208        let fields: Vec<&str> = line.split(',').map(|f| f.trim()).collect();
209        let device = columns
210            .iter()
211            .position(|c| c == "index")
212            .and_then(|i| fields.get(i))
213            .and_then(|v| v.parse::<usize>().ok())
214            .unwrap_or(row_idx);
215
216        for (col, value) in columns.iter().zip(&fields) {
217            if col == "index" {
218                continue;
219            }
220            // "[N/A]" and "[Not Supported]" are normal for some fields on some
221            // cards; skipping keeps them out of the schema entirely rather than
222            // writing nulls forever.
223            if let Ok(v) = value.parse::<f64>() {
224                out.insert(format!("{prefix}.{device}.{col}"), MetricValue::Float(v));
225            }
226        }
227    }
228    out
229}

231/// Extract every numeric leaf, keyed by its dotted path.
232///
233/// Schema-agnostic on purpose — see the module docs. Strings that parse as
234/// numbers count, because `rocm-smi` reports `"GPU use (%)": "42"`.
235fn flatten_json_numbers(value: &serde_json::Value, prefix: &str) -> HashMap<String, MetricValue> {
236    let mut out = HashMap::new();
237    walk_json(value, prefix, &mut out);
238    out
239}
240
241fn walk_json(value: &serde_json::Value, path: &str, out: &mut HashMap<String, MetricValue>) {
242    match value {
243        serde_json::Value::Number(n) => {
244            if let Some(f) = n.as_f64() {
245                out.insert(path.to_string(), MetricValue::Float(f));
246            }
247        }
248        serde_json::Value::String(s) => {
249            if let Some(f) = parse_leading_number(s) {
250                out.insert(path.to_string(), MetricValue::Float(f));
251            }
252        }
253        serde_json::Value::Object(map) => {
254            for (k, v) in map {
255                walk_json(v, &format!("{path}.{}", sanitize(k)), out);
256            }
257        }
258        serde_json::Value::Array(items) => {
259            for (i, v) in items.iter().enumerate() {
260                walk_json(v, &format!("{path}.{i}"), out);
261            }
262        }
263        _ => {}
264    }
265}

267/// Take the leading numeric run of a string.
268///
269/// Vendors glue units on ("42.0c", "155.0W", "0.00%") and quote their numbers,
270/// so the leading run is what carries the value.
271fn parse_leading_number(text: &str) -> Option<f64> {
272    let numeric: String = text
273        .trim()
274        .chars()
275        .take_while(|c| c.is_ascii_digit() || *c == '.' || *c == '-')
276        .collect();
277    (!numeric.is_empty())
278        .then(|| numeric.parse().ok())
279        .flatten()
280}

282/// Parse every markdown-style pipe table in `text`.
283///
284/// The first column identifies the device (`Core ID`, `Device`, `Chip`); every
285/// other column header names a metric. Multiple tables are handled in one pass,
286/// which is what `tpu-info` emits when several `--metric` flags are given.
287///
288/// Rich's surrounding panels use box-drawing characters, not the ASCII `|` these
289/// tables use, so they are skipped without needing to be recognised.
290fn parse_pipe_tables(text: &str, prefix: &str) -> HashMap<String, MetricValue> {
291    let mut out = HashMap::new();
292    let mut header: Option<Vec<String>> = None;
293
294    for line in text.lines() {
295        let trimmed = line.trim();
296        let is_row = trimmed.starts_with('|') && trimmed.ends_with('|') && trimmed.len() > 1;
297        if !is_row {
298            // Any non-row line ends the current table, so the next table's
299            // first row is read as its header rather than as data.
300            header = None;
301            continue;
302        }
303
304        let cells: Vec<String> = trimmed
305            .trim_matches('|')
306            .split('|')
307            .map(|c| c.trim().to_string())
308            .collect();
309
310        // The |---|---| rule under a header carries no data.
311        if cells
312            .iter()
313            .all(|c| !c.is_empty() && c.chars().all(|ch| ch == '-' || ch == ':'))
314        {
315            continue;
316        }
317
318        match &header {
319            None => header = Some(cells),
320            Some(names) => {
321                let Some(device) = cells.first() else {
322                    continue;
323                };
324                let device = sanitize(device);
325                for (name, value) in names.iter().zip(&cells).skip(1) {
326                    // "N/A" is normal: libtpu publishes HBM and duty cycle only
327                    // while a framework holds the TPU. Skipping keeps the column
328                    // out of the schema rather than writing nulls forever.
329                    if let Some(v) = parse_leading_number(value) {
330                        out.insert(
331                            format!("{prefix}.{device}.{}", sanitize(name)),
332                            MetricValue::Float(v),
333                        );
334                    }
335                }
336            }
337        }
338    }
339    out
340}

342fn parse_key_value(text: &str, prefix: &str) -> HashMap<String, MetricValue> {
343    let mut out = HashMap::new();
344    for line in text.lines() {
345        let Some((k, v)) = line.split_once('=').or_else(|| line.split_once(':')) else {
346            continue;
347        };
348        if let Ok(f) = v.trim().parse::<f64>() {
349            out.insert(
350                format!("{prefix}.{}", sanitize(k.trim())),
351                MetricValue::Float(f),
352            );
353        }
354    }
355    out
356}
357
358/// Metric keys become Parquet column names, so keep them boring.
359fn sanitize(key: &str) -> String {
360    key.trim()
361        .chars()
362        .map(|c| {
363            if c.is_alphanumeric() || c == '_' {
364                c
365            } else {
366                '_'
367            }
368        })
369        .collect::<String>()
370        .trim_matches('_')
371        .to_lowercase()
372}

374// ─── Host CPU and memory ──────────────────────────────────────────────────────
375
376/// CPU and memory from `/proc`, which needs no external tool at all.
377#[derive(Default)]
378struct HostProbe {
379    /// Previous cumulative jiffies, since CPU utilisation is a delta.
380    prev_cpu: Option<(u64, u64)>,
381}
382
383impl HostProbe {
384    fn sample(&mut self) -> HashMap<String, MetricValue> {
385        let mut out = HashMap::new();
386
387        if let Ok(stat) = std::fs::read_to_string("/proc/stat") {
388            if let Some(line) = stat.lines().next() {
389                let vals: Vec<u64> = line
390                    .split_whitespace()
391                    .skip(1)
392                    .filter_map(|v| v.parse().ok())
393                    .collect();
394                if vals.len() >= 4 {
395                    let idle = vals[3];
396                    let total: u64 = vals.iter().sum();
397                    // The first sample has no baseline to diff against, so it
398                    // reports nothing rather than a meaningless since-boot mean.
399                    if let Some((prev_total, prev_idle)) = self.prev_cpu {
400                        let dt = total.saturating_sub(prev_total);
401                        let di = idle.saturating_sub(prev_idle);
402                        if dt > 0 {
403                            let used = 100.0 * (dt.saturating_sub(di)) as f64 / dt as f64;
404                            out.insert("cpu.util_pct".to_string(), MetricValue::Float(used));
405                        }
406                    }
407                    self.prev_cpu = Some((total, idle));
408                }
409            }
410        }
411
412        if let Ok(meminfo) = std::fs::read_to_string("/proc/meminfo") {
413            let field = |name: &str| -> Option<f64> {
414                meminfo.lines().find_map(|l| {
415                    let rest = l.strip_prefix(name)?.trim().strip_suffix(" kB")?;
416                    rest.trim()
417                        .parse::<f64>()
418                        .ok()
419                        .map(|kb| kb / 1024.0 / 1024.0)
420                })
421            };
422            if let (Some(total), Some(available)) = (field("MemTotal:"), field("MemAvailable:")) {
423                out.insert("mem.total_gb".to_string(), MetricValue::Float(total));
424                out.insert(
425                    "mem.used_gb".to_string(),
426                    MetricValue::Float(total - available),
427                );
428                if total > 0.0 {
429                    out.insert(
430                        "mem.util_pct".to_string(),
431                        MetricValue::Float(100.0 * (total - available) / total),
432                    );
433                }
434            }
435        }
436
437        // This process's own resident set — usually the number you actually
438        // want when chasing a leak in a training loop.
439        if let Ok(status) = std::fs::read_to_string("/proc/self/status") {
440            if let Some(rss) = status.lines().find_map(|l| {
441                l.strip_prefix("VmRSS:")?
442                    .trim()
443                    .strip_suffix(" kB")?
444                    .trim()
445                    .parse::<f64>()
446                    .ok()
447            }) {
448                out.insert(
449                    "proc.rss_gb".to_string(),
450                    MetricValue::Float(rss / 1024.0 / 1024.0),
451                );
452            }
453        }
454
455        out
456    }
457}

461/// Samples every available probe on demand.
462pub struct SystemSampler {
463    probes: Vec<ProbeSpec>,
464    host: HostProbe,
465}
466
467impl SystemSampler {
468    /// Resolve which probes exist on this machine. Done once, at startup: PATH
469    /// does not change under a running job, and re-checking every tick would
470    /// mean a stat per probe per sample forever.
471    pub fn new(specs: Vec<ProbeSpec>) -> Self {
472        let probes: Vec<ProbeSpec> = specs.into_iter().filter(|p| p.is_available()).collect();
473        if !probes.is_empty() {
474            tracing::debug!(
475                probes = ?probes.iter().map(|p| &p.command).collect::<Vec<_>>(),
476                "system metric probes enabled"
477            );
478        }
479        Self {
480            probes,
481            host: HostProbe::default(),
482        }
483    }
484
485    /// One sample across every probe. Keys collide across probes only if two
486    /// probes share a prefix, which is the caller's choice to make.
487    ///
488    /// Probes run **concurrently**. Each costs a subprocess — `tpu-info` alone is
489    /// ~390ms — so a host with several accelerators would otherwise pay their
490    /// sum on every tick. One thread per probe is right here: they are
491    /// I/O-and-fork bound, there are single digits of them, and they are spawned
492    /// once per sampling interval, not per step.
493    pub fn sample(&mut self) -> HashMap<String, MetricValue> {
494        // The host probe is cheap (a few /proc reads) and holds state across
495        // samples, so it stays inline.
496        let mut out = self.host.sample();
497
498        if self.probes.len() == 1 {
499            out.extend(self.probes[0].sample());
500            return out;
501        }
502
503        let collected: Vec<HashMap<String, MetricValue>> = std::thread::scope(|scope| {
504            let handles: Vec<_> = self
505                .probes
506                .iter()
507                .map(|probe| scope.spawn(move || probe.sample()))
508                .collect();
509            handles
510                .into_iter()
511                // A panicking probe must not take the run's logging with it.
512                .filter_map(|h| h.join().ok())
513                .collect()
514        });
515        for values in collected {
516            out.extend(values);
517        }
518        out
519    }
520
521    /// Names of the external probes actually in use, for reporting.
522    pub fn active_probes(&self) -> Vec<String> {
523        self.probes.iter().map(|p| p.command.clone()).collect()
524    }
525}

613| 0       | 0.00%                  |
614| 1       | 12.50%                 |
615"#;
616
617    #[test]
618    fn tpu_info_pipe_tables_parse() {
619        let out = parse_pipe_tables(TPU_INFO_V6E, "tpu");
620
621        // TensorCore utilisation reports even with no framework attached; the
622        // trailing % is stripped.
623        assert_eq!(out["tpu.0.tensorcore_utilization"].to_string(), "0");
624        assert_eq!(out["tpu.1.tensorcore_utilization"].to_string(), "12.5");
625
626        // N/A means libtpu has nothing to publish yet. Skipped, not zero —
627        // recording 0 GiB of HBM while a job runs would be a lie.
628        assert!(!out.contains_key("tpu.0.duty_cycle"));
629        assert!(
630            out.keys().all(|k| !k.contains("hbm")),
631            "N/A must not be stored: {out:?}"
632        );
633
634        // The Rich warning panel uses box-drawing pipes, not ASCII, so it never
635        // reaches the parser.
636        assert!(out.keys().all(|k| !k.to_lowercase().contains("warning")));
637        assert_eq!(out.len(), 2, "only the two real readings: {out:?}");
638    }

692    #[test]
693    fn host_probe_reports_memory_and_needs_two_samples_for_cpu() {
694        let mut host = HostProbe::default();
695        let first = host.sample();
696        if !cfg!(target_os = "linux") {
697            return;
698        }
699        assert!(first.contains_key("mem.total_gb"));
700        // No baseline yet, so no CPU figure — better than a since-boot average
701        // dressed up as a current reading.
702        assert!(!first.contains_key("cpu.util_pct"));
703
704        // Real elapsed time is required: with a zero jiffy delta there is
705        // genuinely nothing to report, and the sampler correctly says nothing.
706        std::thread::sleep(std::time::Duration::from_millis(60));
707        let second = host.sample();
708        let util = second
709            .get("cpu.util_pct")
710            .and_then(|v| v.to_string().parse::<f64>().ok())
711            .expect("a second sample with elapsed time yields CPU utilisation");
712        assert!(
713            (0.0..=100.0).contains(&util),
714            "utilisation must be a percentage, got {util}"
715        );
716    }
