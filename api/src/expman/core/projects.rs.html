1//! Project sync: projecting an external manifest into the projects layer.
2//!
3//! An expman project can be a **one-way generated projection** of a source that
4//! lives elsewhere — a `studies.yaml` in a thesis repo, say. In that mode expman
5//! is not the authority: it carries only *general information* (display name,
6//! description, tags, a README frontpage) and *experiments information* (which
7//! experiments belong to the project). Hypotheses, task lists and sign-off state
8//! stay in the source.
9//!
10//! The consequence is that a generated project is **overwritten wholesale by the
11//! next sync**. Anything edited through the dashboard in the meantime is lost.
12//! That is why `sync` stamps `generated: true` on the project and writes a
13//! do-not-edit marker into the README: the API refuses edits to such a project
14//! rather than accepting a write it is about to destroy.
15

25/// Marker written at the top of every generated README.
26///
27/// Also the signal `is_generated_readme` looks for, so a README that predates
28/// the `generated` flag is still recognised.
29pub const GENERATED_MARKER: &str =
30    "<!-- expman:generated — do not edit. Regenerated on each sync; edits here are lost. -->";
31
32/// A manifest describing one or more projects to project into the store.
33#[derive(Debug, Clone, Serialize, Deserialize, Default)]
34pub struct ProjectManifest {
35    #[serde(default)]
36    pub projects: Vec<ProjectSpec>,
37}
38
39/// One project's worth of the manifest.
40#[derive(Debug, Clone, Serialize, Deserialize, Default)]
41pub struct ProjectSpec {
42    /// Project id — the directory name under `.projects/`.
43    pub name: String,
44    #[serde(default)]
45    pub display_name: Option<String>,
46    #[serde(default)]
47    pub description: Option<String>,
48    #[serde(default)]
49    pub tags: Vec<String>,
50    /// Human-readable pointer to whatever is authoritative.
51    #[serde(default)]
52    pub generated_from: Option<String>,
53    /// Experiments belonging to this project. Experiments previously assigned
54    /// here but absent from this list are unassigned.
55    #[serde(default)]
56    pub experiments: Vec<String>,
57    /// Optional structured frontpage. Rendered to README.md.
58    #[serde(default)]
59    pub frontpage: Option<Frontpage>,
60    /// Raw README body, used verbatim instead of rendering `frontpage`.
61    #[serde(default)]
62    pub readme: Option<String>,
63}

65/// The structured frontpage of a project, rendered into README.md.
66#[derive(Debug, Clone, Serialize, Deserialize, Default)]
67pub struct Frontpage {
68    #[serde(default)]
69    pub question: Option<String>,
70    #[serde(default)]
71    pub scope: Option<String>,
72    #[serde(default)]
73    pub domains: Vec<String>,
74    #[serde(default)]
75    pub target_venue: Option<String>,
76    #[serde(default)]
77    pub target_date: Option<String>,
78    #[serde(default)]
79    pub status: Option<String>,
80    /// Any further sections, rendered in order as `## <heading>`.
81    #[serde(default)]
82    pub sections: Vec<Section>,
83}

91/// What one project's sync changed. Returned so the caller can report it.
92#[derive(Debug, Clone, Default)]
93pub struct SyncReport {
94    pub project: String,
95    pub assigned: Vec<String>,
96    pub unassigned: Vec<String>,
97    /// Experiments named by the manifest that have no directory in the store.
98    /// Not an error: the project may be declared before its first run.
99    pub missing: Vec<String>,
100}

102/// Load a manifest from a YAML file.
103///
104/// Accepts either the full `{projects: [...]}` form or a bare list of specs.
105pub fn load_manifest(path: &Path) -> Result<ProjectManifest> {
106    let content = std::fs::read_to_string(path)?;
107    // A bare list is the natural thing to write by hand; accept both rather
108    // than failing on a shape that is obviously intended.
109    if let Ok(manifest) = serde_yaml::from_str::<ProjectManifest>(&content) {
110        if !manifest.projects.is_empty() {
111            return Ok(manifest);
112        }
113    }
114    let projects: Vec<ProjectSpec> = serde_yaml::from_str(&content)
115        .map_err(|e| ExpmanError::Other(format!("could not parse manifest: {e}")))?;
116    Ok(ProjectManifest { projects })
117}

119/// Project one spec into the store: metadata, README, and experiment assignment.
120pub fn sync_project(base_dir: &Path, spec: &ProjectSpec) -> Result<SyncReport> {
121    let now = Utc::now();
122
123    // Preserve created_at across syncs — it describes the project, not this run.
124    let previous = storage::load_project_metadata(base_dir, &spec.name).unwrap_or_default();
125    let meta = ProjectMetadata {
126        display_name: spec.display_name.clone(),
127        description: spec.description.clone(),
128        tags: spec.tags.clone(),
129        created_at: previous.created_at.or(Some(now)),
130        generated: true,
131        generated_from: spec.generated_from.clone(),
132        generated_at: Some(now),
133    };
134    storage::save_project_metadata(base_dir, &spec.name, &meta)?;
135
136    let readme = match &spec.readme {
137        Some(raw) => format!("{GENERATED_MARKER}\n\n{}", raw.trim_end()),
138        None => render_readme(spec, &meta),
139    };
140    storage::save_project_readme(base_dir, &spec.name, &readme)?;
141
142    // Reconcile membership: assign everything listed, unassign anything that
143    // was in this project but has dropped out of the manifest.
144    let mut report = SyncReport {
145        project: spec.name.clone(),
146        ..Default::default()
147    };
148    let previously = storage::list_project_experiments(base_dir, &spec.name)?;
149
150    for exp in &spec.experiments {
151        if !base_dir.join(exp).is_dir() {
152            report.missing.push(exp.clone());
153        }
154        storage::set_experiment_project(base_dir, exp, Some(&spec.name))?;
155        if !previously.contains(exp) {
156            report.assigned.push(exp.clone());
157        }
158    }
159
160    for exp in previously {
161        if !spec.experiments.contains(&exp) {
162            storage::set_experiment_project(base_dir, &exp, None)?;
163            report.unassigned.push(exp);
164        }
165    }
166
167    Ok(report)
168}

170/// Sync every project in a manifest.
171pub fn sync_manifest(base_dir: &Path, manifest: &ProjectManifest) -> Result<Vec<SyncReport>> {
172    manifest
173        .projects
174        .iter()
175        .map(|spec| sync_project(base_dir, spec))
176        .collect()
177}

179/// True when a README was written by a sync and will be clobbered by the next one.
180pub fn is_generated_readme(content: &str) -> bool {
181    content.trim_start().starts_with(GENERATED_MARKER)
182}

184/// Render a project's frontpage to Markdown.
185fn render_readme(spec: &ProjectSpec, meta: &ProjectMetadata) -> String {
186    let mut out = String::new();
187    out.push_str(GENERATED_MARKER);
188    out.push_str("\n\n# ");
189    out.push_str(spec.display_name.as_deref().unwrap_or(&spec.name));
190    out.push_str("\n\n");
191
192    if let Some(desc) = &spec.description {
193        out.push_str(desc.trim());
194        out.push_str("\n\n");
195    }
196
197    let fp = spec.frontpage.clone().unwrap_or_default();
198
199    if let Some(question) = &fp.question {
200        out.push_str("## Question\n\n");
201        out.push_str(question.trim());
202        out.push_str("\n\n");
203    }
204    if let Some(scope) = &fp.scope {
205        out.push_str("## Scope\n\n");
206        out.push_str(scope.trim());
207        out.push_str("\n\n");
208    }
209    if !fp.domains.is_empty() {
210        out.push_str("## Domains\n\n");
211        for domain in &fp.domains {
212            out.push_str(&format!("- {domain}\n"));
213        }
214        out.push('\n');
215    }
216
217    // Target and status read as a small fact table rather than prose.
218    let facts: Vec<(&str, String)> = [
219        ("Target venue", fp.target_venue.clone()),
220        ("Target date", fp.target_date.clone()),
221        ("Status", fp.status.clone()),
222        (
223            "Tags",
224            (!spec.tags.is_empty()).then(|| spec.tags.join(", ")),
225        ),
226        (
227            "Experiments",
228            (!spec.experiments.is_empty()).then(|| spec.experiments.len().to_string()),
229        ),
230    ]
231    .into_iter()
232    .filter_map(|(k, v)| v.map(|v| (k, v)))
233    .collect();
234
235    if !facts.is_empty() {
236        out.push_str("| | |\n|---|---|\n");
237        for (key, value) in facts {
238            out.push_str(&format!("| {key} | {value} |\n"));
239        }
240        out.push('\n');
241    }
242
243    for section in &fp.sections {
244        out.push_str(&format!(
245            "## {}\n\n{}\n\n",
246            section.heading,
247            section.body.trim()
248        ));
249    }
250
251    if !spec.experiments.is_empty() {
252        out.push_str("## Experiments\n\n");
253        for exp in &spec.experiments {
254            out.push_str(&format!("- `{exp}`\n"));
255        }
256        out.push('\n');
257    }
258
259    if let Some(source) = &meta.generated_from {
260        out.push_str(&format!(
261            "---\n\nGenerated from {source}{}.\n",
262            meta.generated_at
263                .map(|t| format!(" on {}", t.format("%Y-%m-%d %H:%M UTC")))
264                .unwrap_or_default()
265        ));
266    }
267
268    out
269}
