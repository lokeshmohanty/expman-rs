1//! Provenance: what code and what environment produced a run.
2//!
3//! Written once at run creation into `provenance.yaml`. This is most of what
4//! makes a run reproducible, and it is the thing people notice missing only
5//! months later, when the answer to "which commit was this?" has to be guessed
6//! from a timestamp.
7//!
8//! ## The diff is opt-in, deliberately
9//!
10//! The commit SHA, branch and dirty flag are always captured: they are small,
11//! and they cannot leak anything that is not already in the repository. The
12//! working-tree **diff** is captured only on request, because a dirty tree
13//! routinely contains things that should not be copied into a store you might
14//! later share — an edited `.env`, a pasted API key, an unpublished data path.
15//! Knowing a run was dirty is cheap and safe; knowing exactly how is a choice
16//! the user should make per-run.
17
18use std::path::Path;
19use std::process::Command;
20
21use serde::{Deserialize, Serialize};
22
23/// Everything captured about the code and environment behind a run.
24#[derive(Debug, Clone, Default, Serialize, Deserialize)]
25pub struct Provenance {
26    #[serde(default, skip_serializing_if = "Option::is_none")]
27    pub git: Option<GitInfo>,
28    #[serde(default, skip_serializing_if = "Option::is_none")]
29    pub environment: Option<EnvironmentInfo>,
30    /// Command line the run was launched with, best-effort.
31    #[serde(default, skip_serializing_if = "Option::is_none")]
32    pub command: Option<String>,
33    #[serde(default, skip_serializing_if = "Option::is_none")]
34    pub hostname: Option<String>,
35    /// Scheduler identifiers, when running under one. Lets a run be traced back
36    /// to its scheduler logs, which is where the OOM kill will be recorded.
37    #[serde(default, skip_serializing_if = "std::collections::BTreeMap::is_empty")]
38    pub scheduler: std::collections::BTreeMap<String, String>,
39}
40
41#[derive(Debug, Clone, Default, Serialize, Deserialize)]
42pub struct GitInfo {
43    pub commit: String,
44    #[serde(default, skip_serializing_if = "Option::is_none")]
45    pub branch: Option<String>,
46    /// True when the working tree had uncommitted changes. A run with
47    /// `dirty: true` and no `diff` is *not* reproducible, and says so.
48    pub dirty: bool,
49    #[serde(default, skip_serializing_if = "Option::is_none")]
50    pub remote: Option<String>,
51    /// Working-tree diff, only when explicitly requested.
52    #[serde(default, skip_serializing_if = "Option::is_none")]
53    pub diff: Option<String>,
54    /// Set when a diff was requested but exceeded the size cap.
55    #[serde(default, skip_serializing_if = "Option::is_none")]
56    pub diff_truncated_bytes: Option<usize>,
57}
58
59#[derive(Debug, Clone, Default, Serialize, Deserialize)]
60pub struct EnvironmentInfo {
61    #[serde(default, skip_serializing_if = "Option::is_none")]
62    pub python_version: Option<String>,
63    #[serde(default, skip_serializing_if = "Option::is_none")]
64    pub platform: Option<String>,
65    /// `pip freeze` output. Captured on request — it costs a subprocess and can
66    /// run to hundreds of lines.
67    #[serde(default, skip_serializing_if = "Option::is_none")]
68    pub packages: Option<String>,
69}

71/// A captured diff larger than this is truncated. Big enough for real work,
72/// small enough that a stray notebook checkpoint cannot bloat every run.
73const MAX_DIFF_BYTES: usize = 1024 * 1024;
74
75/// Run a git command in `dir`, returning trimmed stdout on success.
76fn git(dir: &Path, args: &[&str]) -> Option<String> {
77    let output = Command::new("git")
78        .arg("-C")
79        .arg(dir)
80        .args(args)
81        .output()
82        .ok()?;
83    if !output.status.success() {
84        return None;
85    }
86    let text = String::from_utf8_lossy(&output.stdout).trim().to_string();
87    (!text.is_empty()).then_some(text)
88}

90/// Capture git state for the repository containing `dir`.
91///
92/// Returns `None` when `dir` is not in a repository or git is unavailable —
93/// running outside version control is normal, not an error.
94pub fn capture_git(dir: &Path, include_diff: bool) -> Option<GitInfo> {
95    let commit = git(dir, &["rev-parse", "HEAD"])?;
96
97    // `--porcelain` is empty exactly when the tree is clean.
98    let dirty = git(dir, &["status", "--porcelain"]).is_some();
99
100    let branch = git(dir, &["rev-parse", "--abbrev-ref", "HEAD"]).filter(|b| b != "HEAD");
101
102    let mut info = GitInfo {
103        commit,
104        branch,
105        dirty,
106        remote: git(dir, &["config", "--get", "remote.origin.url"]),
107        diff: None,
108        diff_truncated_bytes: None,
109    };
110
111    if include_diff && dirty {
112        // HEAD rather than the index, so staged and unstaged changes both land.
113        if let Some(diff) = git(dir, &["diff", "HEAD"]) {
114            if diff.len() > MAX_DIFF_BYTES {
115                info.diff_truncated_bytes = Some(diff.len());
116                info.diff = Some(diff.chars().take(MAX_DIFF_BYTES).collect());
117            } else {
118                info.diff = Some(diff);
119            }
120        }
121    }
122
123    Some(info)
124}

126/// Scheduler identifiers from the environment.
127///
128/// Only variables that identify *this* job — enough to find it in the
129/// scheduler's own records. Not a dump of the environment, which would sweep up
130/// credentials.
131pub fn capture_scheduler() -> std::collections::BTreeMap<String, String> {
132    const KEYS: &[&str] = &[
133        "SLURM_JOB_ID",
134        "SLURM_ARRAY_JOB_ID",
135        "SLURM_ARRAY_TASK_ID",
136        "SLURM_JOB_NAME",
137        "SLURM_JOB_NODELIST",
138        "SLURM_PROCID",
139        "SLURM_NTASKS",
140        "PBS_JOBID",
141        "LSB_JOBID",
142        "KUBERNETES_SERVICE_HOST",
143    ];
144    KEYS.iter()
145        .filter_map(|k| std::env::var(k).ok().map(|v| (k.to_string(), v)))
146        .collect()
147}

149/// Best-effort hostname, from the environment or `hostname(1)`.
150pub fn capture_hostname() -> Option<String> {
151    std::env::var("HOSTNAME")
152        .ok()
153        .filter(|h| !h.is_empty())
154        .or_else(|| {
155            Command::new("hostname")
156                .output()
157                .ok()
158                .filter(|o| o.status.success())
159                .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string())
160                .filter(|h| !h.is_empty())
161        })
162}
163
164/// The command line this process was started with.
165pub fn capture_command() -> Option<String> {
166    let args: Vec<String> = std::env::args().collect();
167    (!args.is_empty()).then(|| args.join(" "))
168}
169
170impl Provenance {
171    /// Capture everything cheap, plus the diff only if asked.
172    pub fn capture(working_dir: &Path, include_diff: bool) -> Self {
173        Self {
174            git: capture_git(working_dir, include_diff),
175            environment: None,
176            command: capture_command(),
177            hostname: capture_hostname(),
178            scheduler: capture_scheduler(),
179        }
180    }
181
182    /// True when this run could be reconstructed from what was captured.
183    pub fn is_reproducible(&self) -> bool {
184        match &self.git {
185            Some(git) => !git.dirty || git.diff.is_some(),
186            None => false,
187        }
188    }
189}

191#[cfg(test)]
192mod tests {
193    use super::*;
194
195    #[test]
196    fn capture_outside_a_repo_is_none_not_an_error() {
197        let tmp = tempfile::TempDir::new().unwrap();
198        // A bare temp dir is not in a repo (unless /tmp is, which it is not).
199        let prov = Provenance::capture(tmp.path(), false);
200        assert!(prov.git.is_none());
201        // The rest still populates — a run outside git is perfectly normal.
202        assert!(prov.command.is_some());
203    }
204
205    #[test]
206    fn a_clean_repo_is_reproducible_without_a_diff() {
207        let prov = Provenance {
208            git: Some(GitInfo {
209                commit: "abc123".into(),
210                dirty: false,
211                ..Default::default()
212            }),
213            ..Default::default()
214        };
215        assert!(prov.is_reproducible());
216    }
217
218    #[test]
219    fn a_dirty_repo_without_a_diff_is_not_reproducible() {
220        // The point of tracking this: a dirty run is honest about the fact that
221        // its code cannot be recovered.
222        let mut prov = Provenance {
223            git: Some(GitInfo {
224                commit: "abc123".into(),
225                dirty: true,
226                ..Default::default()
227            }),
228            ..Default::default()
229        };
230        assert!(!prov.is_reproducible());
231
232        prov.git.as_mut().unwrap().diff = Some("diff --git ...".into());
233        assert!(prov.is_reproducible());
234    }
235
236    #[test]
237    fn scheduler_capture_only_takes_identifying_keys() {
238        // Not a dump of the environment — a dump would sweep up credentials.
239        unsafe {
240            std::env::set_var("SLURM_JOB_ID", "12345");
241            std::env::set_var("MY_SECRET_TOKEN", "hunter2");
242        }
243        let sched = capture_scheduler();
244        assert_eq!(sched.get("SLURM_JOB_ID").map(String::as_str), Some("12345"));
245        assert!(!sched.contains_key("MY_SECRET_TOKEN"));
246        unsafe {
247            std::env::remove_var("SLURM_JOB_ID");
248            std::env::remove_var("MY_SECRET_TOKEN");
249        }
250    }
251
252    #[test]
253    fn this_repo_is_detected_as_a_git_checkout() {
254        // Runs inside the expman working tree, which is a repo by construction.
255        let here = Path::new(env!("CARGO_MANIFEST_DIR"));
256        let git = capture_git(here, false).expect("expman's own tree is a git repo");
257        assert_eq!(git.commit.len(), 40, "a full SHA: {}", git.commit);
258        assert!(git.diff.is_none(), "diff must stay opt-in");
259    }
