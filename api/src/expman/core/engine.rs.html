1//! Async logging engine: the heart of expman-rs.
2//!
3//! `LoggingEngine::new()` spawns a background tokio task that owns all file handles.
4//! `log_vector()` is a channel send — O(1), never blocks the experiment process.
5//! The background task batches rows and flushes to Parquet periodically.
6

23/// Commands sent to the background logging task.
24enum LogCommand {
25    /// Log a row of vector metrics.
26    Vector(VectorRow),
27    /// Log a single scalar value (replaces if exists).
28    Scalar(HashMap<String, MetricValue>),
29    /// Update the config/params YAML.
30    Params(HashMap<String, serde_yaml::Value>),
31    /// Copy an artifact file into the run's artifacts directory.
32    Artifact(PathBuf),
33    /// Write raw bytes as a media file and record it in the media manifest.
34    Media {
35        tag: String,
36        step: Option<u64>,
37        extension: String,
38        bytes: Vec<u8>,
39    },
40    /// Record a histogram: bin edges and counts for one tag at one step.
41    Histogram {
42        tag: String,
43        step: Option<u64>,
44        edges: Vec<f64>,
45        counts: Vec<u64>,
46    },
47    /// Log a message to the run log file.
48    Log { level: LogLevel, message: String },
49    /// Force flush the current buffer to disk.
50    Flush(oneshot::Sender<Result<()>>),
51    /// Gracefully shut down: flush everything, write final metadata.
52    Shutdown {
53        status: RunStatus,
54        reply: oneshot::Sender<()>,
55    },
56}
57
58#[derive(Debug, Clone, Copy)]
59pub enum LogLevel {
60    Info,
61    Warn,
62    Error,
63}

65/// The non-blocking logging engine.
66///
67/// Internally holds a sender to a tokio mpsc channel. All heavy I/O
68/// happens in a background task on a dedicated tokio runtime thread.
69pub struct LoggingEngine {
70    sender: mpsc::UnboundedSender<LogCommand>,
71    /// Keep the runtime alive as long as the engine exists.
72    _runtime: Arc<Runtime>,
73    config: ExperimentConfig,
74}
75
76impl LoggingEngine {
77    /// Create a new `LoggingEngine` for the given config.
78    ///
79    /// This initializes the run directory, writes initial metadata,
80    /// and spawns the background I/O task.
81    pub fn new(config: ExperimentConfig) -> Result<Self> {
82        // Set up directories
83        let run_dir = config.run_dir();
84        storage::ensure_dir(&run_dir)?;
85        storage::ensure_dir(&run_dir.join("artifacts"))?;
86
87        // Write initial run metadata
88        let now = Utc::now();
89        let meta = RunMetadata {
90            name: config.run_name.clone(),
91            experiment: config.name.clone(),
92            status: RunStatus::Running,
93            started_at: now,
94            heartbeat_at: Some(now),
95            language: Some(config.language.clone()),
96            env_path: config.env_path.clone(),
97            description: config.description.clone(),
98            group: config.group.clone(),
99            rank: config.rank,
100            tags: if config.tags.is_empty() {
101                None
102            } else {
103                Some(config.tags.clone())
104            },
105            ..Default::default()
106        };
107        storage::save_run_metadata(&run_dir, &meta)?;
108
109        // Ensure experiment metadata exists
110        let exp_dir = config.experiment_dir();
111        storage::ensure_dir(&exp_dir)?;
112        let exp_meta_path = exp_dir.join("experiment.yaml");
113        if !exp_meta_path.exists() {
114            storage::save_experiment_metadata(
115                &exp_dir,
116                &crate::core::models::ExperimentMetadata {
117                    project: config.project.clone(),
118                    ..Default::default()
119                },
120            )?;
121        } else if let Some(project) = &config.project {
122            // experiment.yaml is only written when absent, so an explicit
123            // `project=` from a later run would otherwise be silently ignored.
124            // Update just that field and leave the rest of the file alone.
125            let mut existing = storage::load_experiment_metadata(&exp_dir)?;
126            if existing.project.as_deref() != Some(project.as_str()) {
127                existing.project = Some(project.clone());
128                storage::save_experiment_metadata(&exp_dir, &existing)?;
129            }
130        }
131
132        // Capture provenance once, at creation. Cheap enough to be default-on;
133        // the diff stays opt-in (see core::provenance).
134        if config.capture_provenance {
135            let cwd = std::env::current_dir().unwrap_or_else(|_| PathBuf::from("."));
136            let provenance =
137                crate::core::provenance::Provenance::capture(&cwd, config.capture_diff);
138            if let Err(e) = storage::save_yaml(&run_dir.join("provenance.yaml"), &provenance) {
139                error!("Failed to write provenance: {}", e);
140            }
141        }
142
143        // Set up log file
144        let log_path = run_dir.join("run.log");
145
146        // Build dedicated tokio runtime for background I/O
147        let runtime = Arc::new(
148            tokio::runtime::Builder::new_multi_thread()
149                .worker_threads(1)
150                .thread_name("expman-io")
151                .enable_all()
152                .build()
153                .map_err(|e: std::io::Error| ExpmanError::Other(e.to_string()))?,
154        );
155
156        let (sender, receiver) = mpsc::unbounded_channel::<LogCommand>();
157
158        // Spawn background task
159        let flush_rows = config.flush_interval_rows;
160        let flush_ms = config.flush_interval_ms;
161        let heartbeat_secs = config.heartbeat_interval_secs;
162        let system_secs = config.system_metrics_interval_secs;
163        let run_dir_clone = run_dir.clone();
164        runtime.spawn(background_task(
165            receiver,
166            run_dir_clone,
167            log_path,
168            flush_rows,
169            flush_ms,
170            heartbeat_secs,
171            system_secs,
172        ));
173
174        info!(
175            experiment = %config.name,
176            run = %config.run_name,
177            "LoggingEngine initialized"
178        );
179
180        Ok(Self {
181            sender,
182            _runtime: runtime,
183            config,
184        })
185    }

187    /// Log a row of vector metrics. Non-blocking — channel send only.
188    pub fn log_vector(&self, values: HashMap<String, MetricValue>, step: Option<u64>) {
189        let row = VectorRow::new(values, step);
190        // If channel is closed (engine shut down), silently drop.
191        let _ = self.sender.send(LogCommand::Vector(row));
192    }
193
194    /// Log a single scalar value. Non-blocking — replaces existing value for the key.
195    pub fn log_scalar(&self, key: String, value: MetricValue) {
196        let mut map = HashMap::new();
197        map.insert(key, value);
198        let _ = self.sender.send(LogCommand::Scalar(map));
199    }
200
201    /// Log/update experiment parameters (config). Non-blocking.
202    pub fn log_params(&self, params: HashMap<String, serde_yaml::Value>) {
203        let _ = self.sender.send(LogCommand::Params(params));
204    }
205
206    /// Save an artifact file asynchronously. Non-blocking.
207    /// The path is relative to the current working directory for the source,
208    /// and will be preserved as a relative path within the run's artifacts directory.
209    pub fn save_artifact(&self, path: PathBuf) {
210        let _ = self.sender.send(LogCommand::Artifact(path));
211    }
212
213    /// Log a message to the run log. Non-blocking.
214    pub fn log_message(&self, level: LogLevel, message: String) {
215        let _ = self.sender.send(LogCommand::Log { level, message });
216    }
217
218    /// Save an image/audio/video blob under `media/`, indexed by tag and step.
219    pub fn log_media(&self, tag: String, step: Option<u64>, extension: String, bytes: Vec<u8>) {
220        let _ = self.sender.send(LogCommand::Media {
221            tag,
222            step,
223            extension,
224            bytes,
225        });
226    }
227
228    /// Record a pre-binned histogram.
229    pub fn log_histogram(&self, tag: String, step: Option<u64>, edges: Vec<f64>, counts: Vec<u64>) {
230        let _ = self.sender.send(LogCommand::Histogram {
231            tag,
232            step,
233            edges,
234            counts,
235        });
236    }
237
238    /// Force flush the metric buffer to disk. Async — awaits completion.
239    pub async fn flush(&self) -> Result<()> {
240        let (tx, rx) = oneshot::channel();
241        self.sender
242            .send(LogCommand::Flush(tx))
243            .map_err(|_| ExpmanError::ChannelClosed)?;
244        let res: Result<()> = rx.await.map_err(|_| ExpmanError::ChannelClosed)?;
245        res
246    }
247
248    /// Gracefully shut down: flush all pending metrics, write final metadata.
249    /// Blocks until complete. Should be called at experiment end.
250    pub fn close(&self, status: RunStatus) {
251        let (tx, rx) = oneshot::channel();
252        if self
253            .sender
254            .send(LogCommand::Shutdown { status, reply: tx })
255            .is_ok()
256        {
257            // Block current thread until background task confirms shutdown.
258            // We use the runtime's block_on for this.
259            let _ = self._runtime.block_on(rx);
260        }
261    }
262
263    pub fn config(&self) -> &ExperimentConfig {
264        &self.config
265    }
266}

268impl Drop for LoggingEngine {
269    fn drop(&mut self) {
270        // Best-effort graceful shutdown on drop
271        let (tx, rx) = oneshot::channel();
272        if self
273            .sender
274            .send(LogCommand::Shutdown {
275                status: RunStatus::Finished,
276                reply: tx,
277            })
278            .is_ok()
279        {
280            let _ = self
281                ._runtime
282                .block_on(async { tokio::time::timeout(Duration::from_secs(5), rx).await });
283        }
284    }
285}

289async fn background_task(
290    mut receiver: mpsc::UnboundedReceiver<LogCommand>,
291    run_dir: PathBuf,
292    log_path: PathBuf,
293    flush_interval_rows: usize,
294    flush_interval_ms: u64,
295    heartbeat_interval_secs: u64,
296    system_interval_secs: u64,
297) {
298    let mut vector_writer = storage::MetricWriter::new(&run_dir, storage::VECTORS_STEM);
299    let config_path = run_dir.join("config.yaml");
300    let _meta_path = run_dir.join("run.yaml");
301    let artifacts_dir = run_dir.join("artifacts");
302
303    let mut vector_buffer: Vec<VectorRow> = Vec::with_capacity(flush_interval_rows * 2);
304    let mut current_scalars: HashMap<String, MetricValue> = HashMap::new();
305    let mut current_vectors: HashMap<String, MetricValue> = HashMap::new();
306    let mut log_lines: Vec<String> = Vec::new();
307    let mut flush_ticker = interval(Duration::from_millis(flush_interval_ms));
308    flush_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
309
310    // `interval` panics on a zero duration, so clamp and gate the select branch
311    // on the original value instead — 0 means "no heartbeat".
312    let heartbeat_enabled = heartbeat_interval_secs > 0;
313    let mut heartbeat_ticker = interval(Duration::from_secs(heartbeat_interval_secs.max(1)));
314    heartbeat_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
315
316    // System metrics go to their own file so a `gpu.0.util_pct` column never
317    // shows up mixed into the user's own metric namespace.
318    let system_enabled = system_interval_secs > 0;
319    let mut system_sampler = system_enabled.then(|| {
320        crate::core::sysmetrics::SystemSampler::new(crate::core::sysmetrics::ProbeSpec::defaults())
321    });
322    let mut system_writer = storage::MetricWriter::new(&run_dir, storage::SYSTEM_STEM);
323    // Histograms are their own family: they are one row per (tag, step) with a
324    // variable number of bins, which does not belong in the scalar schema.
325    let mut histogram_writer = storage::MetricWriter::new(&run_dir, storage::HISTOGRAM_STEM);
326    let mut system_step: u64 = 0;
327    let mut system_ticker = interval(Duration::from_secs(system_interval_secs.max(1)));
328    system_ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
329
330    let started_at = Utc::now();
331
332    loop {
333        tokio::select! {
334            // Prioritize incoming commands
335            biased;
336
337            cmd = receiver.recv() => {
338                match cmd {
339                    None => {
340                        // Channel closed — flush and exit
341                        flush_vectors(&mut vector_writer, &mut vector_buffer);
342                        flush_logs(&log_path, &mut log_lines);
343                        break;
344                    }
345                    Some(LogCommand::Vector(row)) => {
346                        // Update current vectors with latest values from this row
347                        for (k, v) in row.values.iter() {
348                            current_vectors.insert(k.clone(), v.clone());
349                        }
350                        vector_buffer.push(row);
351                        if vector_buffer.len() >= flush_interval_rows {
352                            flush_vectors(&mut vector_writer, &mut vector_buffer);
353                        }
354                    }
355                    Some(LogCommand::Scalar(scalars)) => {
356                        current_scalars.extend(scalars);
357                    }
358                    Some(LogCommand::Params(params)) => {
359                        handle_params(&config_path, params);
360                    }
361                    Some(LogCommand::Artifact(path)) => {
362                        handle_artifact(&artifacts_dir, path);
363                    }
364                    Some(LogCommand::Media { tag, step, extension, bytes }) => {
365                        handle_media(&run_dir, &tag, step, &extension, &bytes);
366                    }
367                    Some(LogCommand::Histogram { tag, step, edges, counts }) => {
368                        handle_histogram(&mut histogram_writer, &tag, step, &edges, &counts);
369                    }
370                    Some(LogCommand::Log { level, message }) => {
371                        let ts = Utc::now().format("%Y-%m-%dT%H:%M:%S%.3fZ");
372                        let level_str = match level {
373                            LogLevel::Info => "INFO",
374                            LogLevel::Warn => "WARN",
375                            LogLevel::Error => "ERROR",
376                        };
377                        log_lines.push(format!("[{ts}] [{level_str}] {message}"));
378                        if log_lines.len() >= 20 {
379                            flush_logs(&log_path, &mut log_lines);
380                        }
381                    }
382                    Some(LogCommand::Flush(reply)) => {
383                        flush_vectors(&mut vector_writer, &mut vector_buffer);
384                        flush_logs(&log_path, &mut log_lines);
385                        let _ = reply.send(Ok(()));
386                    }
387                    Some(LogCommand::Shutdown { status, reply }) => {
388                        // Final flush, then fold the append-only segments into
389                        // a single Parquet so readers see one tidy file.
390                        flush_vectors(&mut vector_writer, &mut vector_buffer);
391                        flush_logs(&log_path, &mut log_lines);
392                        let _ = vector_writer.finish();
393                        let _ = system_writer.finish();
394                        let _ = histogram_writer.finish();
395                        storage::compact_run(&run_dir);
396
397                        // Update run metadata with final status and latest scalars
398                        let finished_at = Utc::now();
399                        let duration = (finished_at - started_at).num_milliseconds() as f64 / 1000.0;
400
401                        let _ = storage::update_run_metadata(&run_dir, |meta| {
402                            meta.status = status;
403                            meta.finished_at = Some(finished_at);
404                            meta.duration_secs = Some(duration);
405                            if !current_scalars.is_empty() {
406                                meta.scalars = Some(current_scalars.clone());
407                            }
408                            if !current_vectors.is_empty() {
409                                meta.vectors = Some(current_vectors.clone());
410                            }
411                        });
412
413                        let _ = reply.send(());
414                        break;
415                    }
416                }
417            }
418
419            // Periodic flush
420            _ = flush_ticker.tick() => {
421                if !vector_buffer.is_empty() {
422                    flush_vectors(&mut vector_writer, &mut vector_buffer);
423                }
424                if !log_lines.is_empty() {
425                    flush_logs(&log_path, &mut log_lines);
426                }
427
428                // Update metadata with current scalars periodically
429                if !current_scalars.is_empty() || !current_vectors.is_empty() {
430                    let _ = storage::update_run_metadata(&run_dir, |meta| {
431                        meta.scalars = Some(current_scalars.clone());
432                        meta.vectors = Some(current_vectors.clone());
433                    });
434                }
435            }
436
437            // Sample hardware utilisation. Subprocess probes take a few ms, so
438            // this runs on the I/O task and never touches the training loop.
439            _ = system_ticker.tick(), if system_enabled => {
440                if let Some(sampler) = system_sampler.as_mut() {
441                    let values = sampler.sample();
442                    if !values.is_empty() {
443                        let row = VectorRow::new(values, Some(system_step));
444                        system_step += 1;
445                        if let Err(e) = system_writer.append(std::slice::from_ref(&row)) {
446                            error!("Failed to write system metrics: {}", e);
447                        }
448                    }
449                }
450            }
451
452            // Heartbeat: prove the run is still alive so a hard kill is
453            // distinguishable from a legitimately long job. `exp reap` reads this.
454            _ = heartbeat_ticker.tick(), if heartbeat_enabled => {
455                let _ = storage::update_run_metadata(&run_dir, |meta| {
456                    if meta.status == RunStatus::Running {
457                        meta.heartbeat_at = Some(Utc::now());
458                    }
459                });
460            }
461        }
462    }
463}

465fn flush_vectors(writer: &mut storage::MetricWriter, buffer: &mut Vec<VectorRow>) {
466    if buffer.is_empty() {
467        return;
468    }
469    if let Err(e) = writer.append(buffer) {
470        error!("Failed to flush vectors: {}", e);
471    }
472    buffer.clear();
473}
474
475fn flush_logs(path: &std::path::Path, lines: &mut Vec<String>) {
476    if lines.is_empty() {
477        return;
478    }
479    use std::io::Write;
480    match fs::OpenOptions::new().create(true).append(true).open(path) {
481        Ok(mut f) => {
482            for line in lines.iter() {
483                let _ = writeln!(f, "{}", line);
484            }
485        }
486        Err(e) => error!("Failed to write log: {}", e),
487    }
488    lines.clear();
489}

491fn handle_params(config_path: &std::path::Path, new_params: HashMap<String, serde_yaml::Value>) {
492    // Load existing, merge, save
493    let mut existing: HashMap<String, serde_yaml::Value> =
494        storage::load_yaml(config_path).unwrap_or_default();
495    existing.extend(new_params);
496    if let Err(e) = storage::save_yaml(config_path, &existing) {
497        error!("Failed to save params: {}", e);
498    }
499}

501/// Write a media blob and append a line to the media manifest.
502///
503/// Bytes go to a file rather than into Parquet: a Parquet column of image blobs
504/// makes the metrics file unreadable for its actual purpose, and the dashboard
505/// needs a URL it can hand to an `<img>` anyway.
506fn handle_media(
507    run_dir: &std::path::Path,
508    tag: &str,
509    step: Option<u64>,
510    extension: &str,
511    bytes: &[u8],
512) {
513    let media_dir = run_dir.join("media");
514    if let Err(e) = fs::create_dir_all(&media_dir) {
515        error!("Failed to create media dir: {}", e);
516        return;
517    }
518    // Tags are user-supplied and routinely contain '/' ("train/samples"), which
519    // would silently create directories or escape the run.
520    let safe_tag: String = tag
521        .chars()
522        .map(|c| {
523            if c.is_alphanumeric() || c == '-' || c == '_' {
524                c
525            } else {
526                '_'
527            }
528        })
529        .collect();
530    let filename = match step {
531        Some(s) => format!("{safe_tag}-{s:08}.{extension}"),
532        None => format!("{safe_tag}.{extension}"),
533    };
534    let path = media_dir.join(&filename);
535    if let Err(e) = fs::write(&path, bytes) {
536        error!("Failed to write media {}: {}", path.display(), e);
537        return;
538    }
539
540    // A JSONL manifest, appended to: it survives a hard kill mid-run, which a
541    // rewritten index would not.
542    let entry = serde_json::json!({
543        "tag": tag,
544        "step": step,
545        "file": format!("media/{filename}"),
546        "bytes": bytes.len(),
547        "logged_at": Utc::now().to_rfc3339(),
548    });
549    use std::io::Write;
550    match fs::OpenOptions::new()
551        .create(true)
552        .append(true)
553        .open(run_dir.join("media.jsonl"))
554    {
555        Ok(mut f) => {
556            let _ = writeln!(f, "{entry}");
557        }
558        Err(e) => error!("Failed to update media manifest: {}", e),
559    }
560}

562/// Store one histogram as a row: edges and counts as JSON-encoded strings.
563///
564/// Bin counts vary per tag and per step, so a column-per-bin schema would churn
565/// the Parquet schema on every call.
566fn handle_histogram(
567    writer: &mut storage::MetricWriter,
568    tag: &str,
569    step: Option<u64>,
570    edges: &[f64],
571    counts: &[u64],
572) {
573    let mut values: HashMap<String, MetricValue> = HashMap::new();
574    values.insert("tag".to_string(), MetricValue::Text(tag.to_string()));
575    values.insert(
576        "edges".to_string(),
577        MetricValue::Text(serde_json::to_string(edges).unwrap_or_default()),
578    );
579    values.insert(
580        "counts".to_string(),
581        MetricValue::Text(serde_json::to_string(counts).unwrap_or_default()),
582    );
583    values.insert(
584        "total".to_string(),
585        MetricValue::Int(counts.iter().sum::<u64>() as i64),
586    );
587    let row = VectorRow::new(values, step);
588    if let Err(e) = writer.append(std::slice::from_ref(&row)) {
589        error!("Failed to write histogram: {}", e);
590    }
591}

593fn handle_artifact(artifacts_dir: &std::path::Path, path: PathBuf) {
594    // If path is absolute, join() will replace artifacts_dir.
595    // We want to save the file into artifacts_dir, preserving its filename.
596    let dest = if path.is_absolute() {
597        if let Some(filename) = path.file_name() {
598            artifacts_dir.join(filename)
599        } else {
600            error!("Invalid artifact path: {}", path.display());
601            return;
602        }
603    } else {
604        artifacts_dir.join(&path)
605    };
606
607    if let Some(parent) = dest.parent() {
608        if let Err(e) = fs::create_dir_all(parent) {
609            error!("Failed to create artifact dir: {}", e);
610            return;
611        }
612    }
613    if let Err(e) = fs::copy(&path, &dest) {
614        error!(
615            "Failed to copy artifact {} -> {}: {}",
616            path.display(),
617            dest.display(),
618            e
619        );
620    }
